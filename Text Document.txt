
ПРИЛОЖЕНИЕ «БЛАНК И ТАХО» — ИСХОДНЫЙ КОД
==========================================

Что это: приложение заполняет бланк «Spesen und Arbeitszeiten» и листы «Tageskontrollblatt»
(тахограф, 4 дня на лист) по месяцам и собирает всё в один PDF. Работает без сервера и без
интернета: данные хранятся в localStorage, PDF рисуется на canvas в браузере.

Стек: React 19 + TypeScript, react-router-dom 7, @tanstack/react-table 8, lucide-react, Tailwind CSS 4.

Правила, которые заложены в расчёт:
- Бланк: старт 5:30–8:00 (шаг 30 мин), Netto Arb.Zeit = Fahrzeit + 1 час, пауза 1 час,
  часы вождения за месяц делятся по рабочим дням (не больше 9 ч в день), Spesen 14 € (пн, пт) или 28 €.
- Tacho: день из клеток по 15 минут. Строка 1 — вождение, 2 — отдых и перерывы, 3 — прочая работа.
  День начинается и заканчивается ездой (сначала 15/30/60 минут, потом работа). Редко (1 день из 8)
  начинается или заканчивается работой. Вождение делится на 2–3 периода.
  Вождение без перерыва не больше 4:30. Перерыв: 45 минут или 15 + 30. Вождение в день не больше 10 ч.
  Выходной = весь день отдых, пробег 0. Пробег за месяц = конечный километраж − начальный,
  делится по дням, одинаковых чисел нет.

Как собрать: создайте Vite-проект React + TypeScript, положите файлы по путям ниже
(client/lib/..., client/pages/..., client/App.tsx).

ВАЖНО: компоненты интерфейса из папки client/components (Button, Card, Input, Label, Select,
DataTable, Stat, Badge, DetailList, Dialog, PageHeader, Nav, Toast) принадлежат платформе Glide и
здесь не приведены. Это обычные кнопки, поля, таблицы и диалоги: замените их своими или
библиотекой (например, shadcn/ui). Вся логика находится в client/lib и от них не зависит.
Файл client/lib/offline.ts и server.ts нужны только для режима без интернета (service worker).

// Ручные правки одного дня. Всё, чего здесь нет, считается автоматически.
export interface DayEdit {
  von?: number; // минуты от полуночи
  drive?: number; // Netto Fahrzeit, ч
  pause?: number; // Pause, ч
  zeit?: string;
  spesen?: string;
  start?: string;
}

export interface BlankInput {
  name: string;
  plate: string;
  year: string; // YYYY
  monthNum: string; // "01".."12"
  monthDrive: string; // сумма Netto Fahrzeit за месяц, ч
  seed: number; // меняется кнопкой «Перемешать»
  startZip: string; // индекс старта первого дня
  stops: Record<string, string>; // индекс конца дня (Stop) по дате YYYY-MM-DD
  off: Record<string, boolean>; // будний день, сделанный нерабочим
  on: Record<string, boolean>; // выходной (сб, вс), на который выходят работать
  edits: Record<string, DayEdit>; // ручные правки по дате
  odoStart: string; // километраж на начало месяца (Tageskontrollblatt)
  odoEnd: string; // километраж на конец месяца
}

export function parseHours(s: string): number {
  const t = s.trim().replace(",", ".");
  if (!t) return 0;
  if (t.includes(":")) {
    const [h, m] = t.split(":");
    const v = Number(h) + Number(m || 0) / 60;
    return Number.isFinite(v) ? v : 0;
  }
  const v = Number(t);
  return Number.isFinite(v) ? v : 0;
}

// "6:30", "06.30", "630" → минуты; иначе null
function parseTime(s: string): number | null {
  const t = s.trim();
  let h: number;
  let m: number;
  const a = /^(\d{1,2})[:.,](\d{2})$/.exec(t);
  const b = /^(\d{1,2})(\d{2})$/.exec(t);
  if (a) [h, m] = [Number(a[1]), Number(a[2])];
  else if (b) [h, m] = [Number(b[1]), Number(b[2])];
  else if (/^\d{1,2}$/.test(t)) [h, m] = [Number(t), 0];
  else return null;
  if (h > 23 || m > 59) return null;
  return h * 60 + m;
}

function fmtTime(minutes: number): string {
  const t = ((Math.round(minutes) % 1440) + 1440) % 1440;
  const h = String(Math.floor(t / 60)).padStart(2, "0");
  const mm = String(t % 60).padStart(2, "0");
  return `${h}:${mm}`;
}

export function fmtNum(n: number): string {
  return String(Math.round(n * 100) / 100).replace(".", ",");
}

// Часы в формате 7:30, 10:00
export function fmtHM(hours: number): string {
  const total = Math.max(0, Math.round(hours * 60));
  return `${Math.floor(total / 60)}:${String(total % 60).padStart(2, "0")}`;
}

// Детерминированный генератор случайных чисел: при одном seed результат одинаковый.
export function rng(seed: number) {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

// Делит часы на n рабочих дней: у каждого дня своё значение с шагом
// в полчаса, вокруг среднего (±1 ч). Сумма сходится точно.
export function splitDrive(totalHours: number, n: number, seed: number): number[] {
  if (n <= 0) return [];
  const units = Math.max(0, Math.round(totalHours * 2)); // полчаса = 1
  const base = Math.floor(units / n);
  const u = Array.from({ length: n }, () => base);
  const rand = rng(seed * 31 + n * 7 + units);
  const order = Array.from({ length: n }, (_, i) => i).sort(() => rand() - 0.5);
  for (let k = 0; k < units - base * n; k++) u[order[k]] += 1;
  const lo = Math.max(0, base - 2);
  // не больше 9 ч в день: при часовой паузе больше вести по закону нельзя
  const hi = Math.max(base, Math.min(base + 3, 18));
  for (let k = 0; k < n * 4; k++) {
    const i = Math.floor(rand() * n);
    const j = Math.floor(rand() * n);
    if (i !== j && u[j] - 1 >= lo && u[i] + 1 <= hi) {
      u[i] += 1;
      u[j] -= 1;
    }
  }
  return u.map((x) => x / 2);
}

// Старт утром: от 5:30 до 8:00 с шагом в полчаса.
function startMinutes(rand: () => number): number {
  return 330 + 30 * Math.floor(rand() * 6);
}

const isoOf = (input: BlankInput, d: number) =>
  `${input.year}-${input.monthNum}-${String(d).padStart(2, "0")}`;

export function computeBlank(input: BlankInput): { rows: BlankRow[]; totals: BlankTotals } {
  const y = Number(input.year);
  const m = Number(input.monthNum);
  const totals: BlankTotals = { spesen: 0, arb: 0, fahr: 0 };
  if (!y || !m) return { rows: [], totals };

  const days = new Date(y, m, 0).getDate();
  const isWork = (d: number) => {
    const wd = new Date(y, m - 1, d).getDay();
    const iso = isoOf(input, d);
    return wd >= 1 && wd <= 5 ? !input.off[iso] : Boolean(input.on?.[iso]);
  };

  // Часы, заданные вручную, остаются как есть; остаток месяца делится
  // между остальными рабочими днями.
  let locked = 0;
  let free = 0;
  for (let d = 1; d <= days; d++) {
    if (!isWork(d)) continue;
    const e = input.edits[isoOf(input, d)];
    if (e?.drive !== undefined) locked += e.drive;
    else free++;
  }
  const drives = splitDrive(parseHours(input.monthDrive) - locked, free, input.seed);

  const startRand = rng(input.seed * 17 + 5);
  let wi = 0;
  const rows: BlankRow[] = [];
  let currentZip = input.startZip.trim();

  for (let d = 1; d <= days; d++) {
    const iso = isoOf(input, d);
    const weekday = new Date(y, m - 1, d).getDay();
    const work = isWork(d);
    // старт тянется для каждого календарного дня: отключение одного дня
    // не сдвигает время старта у остальных
    const randomVon = startMinutes(startRand);
    const edit = input.edits[iso];
    const row: BlankRow = {
      iso, day: d, weekday, work,
      edited: Boolean(edit) || Boolean(input.stops[iso]),
      vonMin: 0, driveH: 0, pauseH: 0,
      von: "", bis: "", zeit: "", spesen: "", start: "", stop: "",
      pvon: "", pause: "", pbis: "", arb: "", fahr: "",
    };
    if (work) {
      const e = edit ?? {};
      const drive = e.drive ?? drives[wi++];
      const von = e.von ?? randomVon;
      const pause = e.pause ?? 1;
      const arb = drive + 1; // Netto Arb.Zeit = Fahrzeit + 1 час
      // bis = von + Arb.Zeit + Pause (при Pause 1: von + Fahrzeit + 2 часа)
      const bis = fmtTime(von + Math.round((arb + pause) * 60));
      const vonStr = fmtTime(von);
      const spesen = e.spesen ?? String(weekday === 1 || weekday === 5 ? 14 : 28);
      const stop = (input.stops[iso] ?? "").trim();
      row.vonMin = von;
      row.driveH = drive;
      row.pauseH = pause;
      row.von = vonStr;
      row.bis = bis;
      row.zeit = e.zeit ?? "24";
      row.spesen = spesen;
      row.start = e.start ?? currentZip;
      row.stop = stop;
      row.pvon = vonStr; // правый von копируется из левого
      row.pause = fmtNum(pause);
      row.pbis = bis;
      row.arb = fmtHM(arb);
      row.fahr = fmtHM(drive);
      if (stop) currentZip = stop; // Stop дня переходит в Start следующего
      totals.spesen += parseHours(spesen);
      totals.arb += arb;
      totals.fahr += drive;
    }
    rows.push(row);
  }
  return { rows, totals };
}

// Ручная правка одного поля. Остальные поля дня и остальные дни
// пересчитываются в computeBlank. Пустое значение возвращает автоматику.
export function applyEdit(
  input: BlankInput,
  row: BlankRow,
  field: FieldKey,
  raw: string,
): BlankInput {
  if (!row.work) return input;
  const value = raw.trim();
  const e: DayEdit = { ...(input.edits[row.iso] ?? {}) };
  const stops = { ...input.stops };

  switch (field) {
    case "von":
    case "pvon": {
      if (!value) delete e.von;
      else {
        const t = parseTime(value);
        if (t === null) return input;
        e.von = t;
      }
      break;
    }
    case "bis":
    case "pbis": {
      if (!value) delete e.drive;
      else {
        const t = parseTime(value);
        if (t === null) return input;
        const span = ((((t - row.vonMin) % 1440) + 1440) % 1440) / 60;
        e.drive = Math.max(0, Math.round((span - 1 - row.pauseH) * 100) / 100);
      }
      break;
    }
    case "arb":
      if (!value) delete e.drive;
      else e.drive = Math.max(0, parseHours(value) - 1);
      break;
    case "fahr":
      if (!value) delete e.drive;
      else e.drive = Math.max(0, parseHours(value));
      break;
    case "pause":
      if (!value) delete e.pause;
      else e.pause = Math.max(0, parseHours(value));
      break;
    case "zeit":
      if (!value) delete e.zeit;
      else e.zeit = value;
      break;
    case "spesen":
      if (!value) delete e.spesen;
      else e.spesen = value;
      break;
    case "start":
      if (!value) delete e.start;
      else e.start = value;
      break;
    case "stop":
      if (!value) delete stops[row.iso];
      else stops[row.iso] = value;
      break;
  }

  const edits = { ...input.edits };
  if (Object.keys(e).length) edits[row.iso] = e;
  else delete edits[row.iso];
  return { ...input, edits, stops };
}

// Вернуть день к автоматическим значениям.
export function resetDay(input: BlankInput, iso: string): BlankInput {
  const edits = { ...input.edits };
  const stops = { ...input.stops };
  delete edits[iso];
  delete stops[iso];
  return { ...input, edits, stops };
}

import { computeBlank, rng, type BlankInput } from "./blank";

// Tageskontrollblatt: каждый день = 96 клеток по 15 минут.
// Строка 1 — Lenkzeiten (вождение), 2 — Ruhezeiten/перерывы, 3 — прочая работа.
export type Kind = "drive" | "rest" | "work";

export const UNITS = 96;
const MAX_RUN = 18; // 4:30 вождения до обязательного перерыва
const MAX_DAY_DRIVE = 40; // 10:00 вождения в день
const WORK_HOUR = 4; // Netto Arb.Zeit = Fahrzeit + 1 час

type Tok = { t: "D" | "P" | "W"; u: number };

export interface TachoDay {
  iso: string;
  day: number;
  weekday: number;
  work: boolean;
  dateLabel: string;
  from: string;
  to: string;
  kmStart: string;
  kmEnd: string;
  km: number | null; // null — пробег неизвестен (не введён километраж конца месяца)
  cells: Kind[];
  vonU: number;
  endU: number;
  driveU: number;
  pauseU: number;
  drives: number[]; // периоды вождения, в клетках
  maxRun: number; // самое долгое вождение до перерыва, в клетках
  warnings: string[];
}

export interface TachoResult {
  firstName: string;
  lastName: string;
  plate: string;
  days: TachoDay[];
  driveU: number;
  km: number;
  warningCount: number;
}

const WD_DE = ["So", "Mo", "Di", "Mi", "Do", "Fr", "Sa"];

export function unitTime(u: number): string {
  const m = Math.min(1440, u * 15);
  return `${String(Math.floor(m / 60)).padStart(2, "0")}:${String(m % 60).padStart(2, "0")}`;
}

export function unitHM(u: number): string {
  const m = u * 15;
  return `${Math.floor(m / 60)}:${String(m % 60).padStart(2, "0")}`;
}

// Делит total на n частей, каждая не меньше min.
function split(total: number, n: number, min: number, rand: () => number): number[] {
  if (n <= 0) return [];
  const parts = Array.from({ length: n }, () => min);
  let left = total - min * n;
  if (left < 0) {
    const base = Math.floor(total / n);
    const even = Array.from({ length: n }, () => base);
    for (let i = 0; i < total - base * n; i++) even[i]++;
    return even;
  }
  while (left-- > 0) parts[Math.floor(rand() * n)]++;
  return parts;
}

// Правило перерыва: 45 минут сразу или 15 + потом 30. После такого перерыва
// счётчик вождения обнуляется. Прочая работа перерывом не считается.
function analyze(seq: Tok[]) {
  let run = 0;
  let maxRun = 0;
  let pending = false;
  let ok = true;
  for (const s of seq) {
    if (s.t === "D") {
      run += s.u;
      maxRun = Math.max(maxRun, run);
      if (run > MAX_RUN) ok = false;
    } else if (s.t === "P") {
      if (s.u >= 3) {
        run = 0;
        pending = false;
      } else if (pending && s.u >= 2) {
        run = 0;
        pending = false;
      } else if (s.u >= 1) {
        pending = true;
      }
    }
  }
  return { ok, maxRun };
}

// Запасной вариант, когда вождения почти нет: работа по краям.
function buildFlat(driveU: number, pauseU: number, workU: number, rand: () => number): { seq: Tok[]; drives: number[] } {
  // 2 или 3 периода вождения; один — только если вождения совсем мало
  let n = driveU === 0 ? 0 : driveU < 4 ? 1 : Math.max(2, Math.ceil(driveU / MAX_RUN));
  if (n === 2 && driveU >= 24 && rand() < 0.4) n = 3;
  n = Math.min(n, pauseU + 1);
  if (driveU > 0 && n < 1) n = 1;

  const pauses = n > 1 ? split(pauseU, n - 1, 1, rand) : [];
  const blocks = n > 0 ? split(driveU, n, Math.min(4, Math.floor(driveU / n)), rand) : [];

  // час прочей работы: 2–3 куска по 15 минут и кратно
  let wantW = workU > 0 ? Math.min(workU, n >= 3 || rand() < 0.35 ? 3 : 2) : 0;
  if (n < 2) wantW = Math.min(wantW, 2);
  const wparts = wantW ? split(workU, wantW, 1, rand) : [];
  const wa = wparts[0] ?? 0;
  const we = wparts.length > 1 ? wparts[wparts.length - 1] : 0;
  const wm = wparts.length > 2 ? wparts[1] : 0;
  const mi = n > 1 ? Math.floor(rand() * (n - 1)) : 0;
  const before = rand() < 0.5;

  const seq: Tok[] = [];
  const add = (t: Tok["t"], u: number) => u > 0 && seq.push({ t, u });
  add("W", wa);
  for (let i = 0; i < n; i++) {
    add("D", blocks[i]);
    if (i < n - 1) {
      if (wm && i === mi && before) add("W", wm);
      add("P", pauses[i]);
      if (wm && i === mi && !before) add("W", wm);
    }
  }
  if (n <= 1) add("P", pauseU);
  add("W", we);
  return { seq, drives: blocks };
}

function shuffle<T>(arr: T[], rand: () => number): T[] {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(rand() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}

// День начинается и заканчивается ездой: сначала короткий отрезок (15, 30
// или 60 минут), потом работа. Работа и перерывы стоят между отрезками
// вождения. Редко (примерно 1 день из 8) день начинается или кончается работой.
function build(
  driveU: number,
  pauseU: number,
  workU: number,
  rand: () => number,
  attempt: number,
): { seq: Tok[]; drives: number[] } {
  if (driveU < 2) return buildFlat(driveU, pauseU, workU, rand);

  let n = Math.max(2, Math.ceil(driveU / MAX_RUN));
  if (driveU >= 12 && rand() < 0.4) n++;
  if (attempt > 150) n++;
  n = Math.min(n, 4, driveU);

  const firstOpts = [1, 2, 4, 4].filter((x) => x <= driveU - (n - 1));
  const firstLen = firstOpts.length ? firstOpts[Math.floor(rand() * firstOpts.length)] : 1;
  const rem = driveU - firstLen;
  const rest = split(rem, n - 1, Math.min(4, Math.floor(rem / (n - 1))), rand);
  const blocks = [firstLen, ...rest];

  // кусочки часа прочей работы: 2–3 по 15 минут и кратно
  const k = workU >= 3 && rand() < 0.45 ? 3 : Math.min(2, workU);
  const parts = k > 0 ? split(workU, k, 1, rand) : [];
  let startW = 0;
  let endW = 0;
  if (parts.length >= 2) {
    const r = rand();
    if (r < 0.06) startW = parts.pop()!;
    else if (r < 0.12) endW = parts.pop()!;
  }

  const gaps: Tok[][] = Array.from({ length: n - 1 }, () => []);
  if (parts.length) gaps[0].push({ t: "W", u: parts[0] });
  for (let i = 1; i < parts.length; i++) gaps[Math.floor(rand() * gaps.length)].push({ t: "W", u: parts[i] });

  if (pauseU > 0) {
    const m = 1 + Math.floor(rand() * Math.min(n - 1, pauseU));
    const amounts = split(pauseU, m, 1, rand);
    const where = shuffle(Array.from({ length: n - 1 }, (_, i) => i), rand).slice(0, m);
    where.forEach((g, i) => gaps[g].push({ t: "P", u: amounts[i] }));
  }

  const seq: Tok[] = [];
  const add = (tok: Tok) => {
    const prev = seq[seq.length - 1];
    if (prev && prev.t === tok.t) prev.u += tok.u;
    else seq.push({ ...tok });
  };
  if (startW) add({ t: "W", u: startW });
  blocks.forEach((b, i) => {
    add({ t: "D", u: b });
    if (i < n - 1) {
      const items = gaps[i];
      const head = i === 0 && parts.length ? items.slice(0, 1) : [];
      const tail = shuffle(items.slice(head.length), rand);
      [...head, ...tail].forEach(add);
    }
  });
  if (endW) add({ t: "W", u: endW });
  return { seq, drives: blocks };
}

function plan(driveU: number, pauseU: number, workU: number, rand: () => number) {
  let last = build(driveU, pauseU, workU, rand, 0);
  let a = analyze(last.seq);
  for (let i = 1; i < 400 && !a.ok; i++) {
    last = build(driveU, pauseU, workU, rand, i);
    a = analyze(last.seq);
  }
  return { ...last, ...a };
}

const digits = (s: string) => Number(s.replace(/[^\d]/g, ""));

// Делит пробег месяца по дням с вождением: примерно пропорционально часам,
// но у каждого дня своё число, одинаковых в месяце нет. Сумма точная.
function distributeKm(
  rows: { iso: string; work: boolean; driveH: number }[],
  total: number,
  seed: number,
): Record<string, number> {
  const list = rows.filter((r) => r.work && r.driveH > 0);
  if (total <= 0 || !list.length) return {};
  const rand = rng(seed * 977 + 11);
  const weights = list.map((r) => r.driveH * (0.8 + 0.4 * rand()));
  const sumW = weights.reduce((s, w) => s + w, 0);
  const vals = weights.map((w) => Math.max(1, Math.round((total * w) / sumW)));

  const taken = new Set<number>();
  vals.forEach((v, i) => {
    let nv = v;
    for (let d = 1; taken.has(nv); d++) nv = d % 2 ? v + (d + 1) / 2 : Math.max(1, v - d / 2);
    vals[i] = nv;
    taken.add(nv);
  });

  let diff = total - vals.reduce((s, v) => s + v, 0);
  const step = diff > 0 ? 1 : -1;
  for (let guard = 0; diff !== 0 && guard < 200000; guard++) {
    const i = Math.floor(rand() * vals.length);
    const nv = vals[i] + step;
    if (nv < 1 || taken.has(nv)) continue;
    taken.delete(vals[i]);
    taken.add(nv);
    vals[i] = nv;
    diff -= step;
  }
  const out: Record<string, number> = {};
  list.forEach((r, i) => (out[r.iso] = vals[i]));
  return out;
}

export function computeTacho(input: BlankInput): TachoResult {
  const { rows } = computeBlank(input);
  const [firstName = "", ...rest] = input.name.trim().split(/\s+/).filter(Boolean);
  const lastName = rest.join(" ");

  const startText = (input.odoStart ?? "").replace(/[^\d]/g, "");
  const endText = (input.odoEnd ?? "").replace(/[^\d]/g, "");
  let odo = startText ? digits(startText) : NaN;
  const hasOdo = Number.isFinite(odo);
  const monthKm = startText && endText ? digits(endText) - digits(startText) : 0;
  const kmByDay = distributeKm(rows, monthKm, input.seed);

  let zip = input.startZip.trim();
  const days: TachoDay[] = [];

  for (const r of rows) {
    const day: TachoDay = {
      iso: r.iso,
      day: r.day,
      weekday: r.weekday,
      work: r.work,
      dateLabel: `${WD_DE[r.weekday]} ${String(r.day).padStart(2, "0")}.${input.monthNum}.${input.year}`,
      from: zip,
      to: zip,
      kmStart: hasOdo ? String(odo) : "",
      kmEnd: hasOdo ? String(odo) : "",
      km: 0,
      cells: Array.from({ length: UNITS }, () => "rest" as Kind),
      vonU: 0,
      endU: 0,
      driveU: 0,
      pauseU: 0,
      drives: [],
      maxRun: 0,
      warnings: [],
    };

    if (r.work) {
      const km = kmByDay[r.iso];
      day.km = km ?? null;
      if (hasOdo && km !== undefined) {
        day.kmEnd = String(odo + km);
        odo += km;
      } else {
        day.kmEnd = "";
        day.kmStart = hasOdo ? String(odo) : "";
      }
      day.from = r.start || zip;
      day.to = r.stop || day.from;
      zip = day.to;

      const vonU = Math.round(r.vonMin / 15);
      const driveU = Math.round(r.driveH * 4);
      const pauseU = Math.round(r.pauseH * 4);
      const rand = rng(input.seed * 131 + r.day * 7919 + 17);
      const p = plan(driveU, pauseU, WORK_HOUR, rand);

      let pos = vonU;
      for (const s of p.seq) {
        const kind: Kind = s.t === "D" ? "drive" : s.t === "P" ? "rest" : "work";
        for (let k = 0; k < s.u; k++, pos++) if (pos < UNITS) day.cells[pos] = kind;
      }
      Object.assign(day, { vonU, endU: pos, driveU, pauseU, drives: p.drives, maxRun: p.maxRun });

      if (driveU > MAX_DAY_DRIVE) day.warnings.push(`вождение ${unitHM(driveU)} — больше 10 ч в день`);
      if (!p.ok) day.warnings.push("нет положенного перерыва (45 мин после каждых 4:30 вождения) — увеличьте Pause в бланке");
      if (pos > UNITS) day.warnings.push("смена выходит за полночь, лист обрезан");
    }
    days.push(day);
  }

  return {
    firstName,
    lastName,
    plate: input.plate.trim(),
    days,
    driveU: days.reduce((s, d) => s + d.driveU, 0),
    km: days.reduce((s, d) => s + (d.km ?? 0), 0),
    warningCount: days.reduce((s, d) => s + d.warnings.length, 0),
  };
}

    box(LEFT, y1 + rh, 130, rh);

      box(RX + LAB, y1 + k * rh, 60 - LAB, rh);
      text(label, RX + 1, mid(k), 2.4, { bold: true });
      text(value, RX + LAB + 1.5, mid(k), 3.2);

      // закрашенные клетки

        const row = kind === "drive" ? 0 : kind === "rest" ? 1 : 2;
        ctx.fillStyle = INK;
        ctx.fillRect(GX * S + c * cell * S, (gy + row * rowH) * S, cell * S, rowH * S);

        icon(r, LEFT + 4, y + rowH / 2);
        if (half === 0) text(`${r + 1}*`, LEFT - 1, y + rowH / 2, 2.6, { align: "right" });
      }
      box(GX, gy, GW, 3 * rowH, 0.4);
    }
  }

  function icon(r: number, cx: number, cy: number) {
    ctx.strokeStyle = INK;
    lw(0.25);
    if (r === 0) {
      ctx.beginPath();
      ctx.arc(cx * S, cy * S, 1.8 * S, 0, Math.PI * 2);
      ctx.stroke();
      line(cx - 1.8, cy, cx + 1.8, cy, 0.25);
      line(cx, cy - 1.8, cx, cy + 0.2, 0.25);
    } else if (r === 1) {
      ctx.fillStyle = INK;
      ctx.fillRect((cx - 3) * S, (cy + 0.3) * S, 6 * S, 1.1 * S);
      ctx.fillRect((cx - 3) * S, (cy - 1.4) * S, 1 * S, 2.8 * S);
      ctx.fillRect((cx - 1.8) * S, (cy - 0.8) * S, 2 * S, 1 * S);
    } else {
      ctx.strokeRect((cx - 1.8) * S, (cy - 1.6) * S, 3.6 * S, 3.2 * S);
      line(cx - 1.8, cy + 1.6, cx + 1.8, cy - 1.6, 0.25);
    }
  }

  return canvas;
}

// PDF из нескольких листов A4, по одной картинке на лист, без внешних библиотек.
export async function canvasesToPdf(canvases: HTMLCanvasElement[]): Promise<Blob> {
  const jpgs = await Promise.all(
    canvases.map(async (canvas) => {
      const blob = await new Promise<Blob>((resolve, reject) =>
        canvas.toBlob((b) => (b ? resolve(b) : reject(new Error("canvas"))), "image/jpeg", 0.93),
      );
      return new Uint8Array(await blob.arrayBuffer());
    }),
  );
  const enc = new TextEncoder();
  const chunks: Uint8Array[] = [];
  const offsets: number[] = [];
  let offset = 0;
  const push = (d: string | Uint8Array) => {
    const b = typeof d === "string" ? enc.encode(d) : d;
    chunks.push(b);
    offset += b.length;
  };
  const obj = (n: number, body: string) => {
    offsets[n] = offset;
    push(`${n} 0 obj\n${body}\nendobj\n`);
  };

  const n = canvases.length;
  const kids = canvases.map((_, i) => `${3 + 3 * i} 0 R`).join(" ");
  push("%PDF-1.4\n");
  obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
  obj(2, `<< /Type /Pages /Kids [${kids}] /Count ${n} >>`);
  canvases.forEach((canvas, i) => {
    const pageId = 3 + 3 * i;
    const imgId = pageId + 1;
    const contId = pageId + 2;
    obj(pageId, `<< /Type /Page /Parent 2 0 R /MediaBox [0 0 595.28 841.89] /Resources << /XObject << /Im${i} ${imgId} 0 R >> >> /Contents ${contId} 0 R >>`);
    offsets[imgId] = offset;
    push(
      `${imgId} 0 obj\n<< /Type /XObject /Subtype /Image /Width ${canvas.width} /Height ${canvas.height} /ColorSpace /DeviceRGB /BitsPerComponent 8 /Filter /DCTDecode /Length ${jpgs[i].length} >>\nstream\n`,
    );
    push(jpgs[i]);
    push("\nendstream\nendobj\n");
    const content = `q 595.28 0 0 841.89 0 0 cm /Im${i} Do Q`;
    obj(contId, `<< /Length ${content.length} >>\nstream\n${content}\nendstream`);
  });
  const size = 3 + 3 * n;
  const xref = offset;
  let table = `xref\n0 ${size}\n0000000000 65535 f \n`;
  for (let k = 1; k < size; k++) table += `${String(offsets[k]).padStart(10, "0")} 00000 n \n`;
  push(`${table}trailer\n<< /Size ${size} /Root 1 0 R >>\nstartxref\n${xref}\n%%EOF\n`);
  return new Blob(chunks as BlobPart[], { type: "application/pdf" });
}

// Один PDF: первая страница — Spesen und Arbeitszeiten, дальше листы Tageskontrollblatt.
export async function downloadCombinedPdf(form: BlankInput) {
  const { rows, totals } = computeBlank(form);
  const tacho = computeTacho(form);
  const canvases = [
    drawBlank(form, rows, totals),
    ...Array.from({ length: pageCount(tacho.days.length) }, (_, p) => drawTachoPage(tacho, p, 8)),
  ];
  const pdf = await canvasesToPdf(canvases);
  const who = (form.name.trim() || "Spesen").replace(/\s+/g, "_");
  downloadBlob(pdf, `Spesen_Tacho_${who}_${form.year}-${form.monthNum}.pdf`);
}

  return {

    year: String(now.getFullYear()),
    monthNum: String(now.getMonth() + 1).padStart(2, "0"),

    off: {},
    on: {},

    odoStart: "",
    odoEnd: "",
  };
}

function initialState(): BlankInput {
  const base = emptyState();
  try {
    const saved = localStorage.getItem(STORAGE_KEY);
    if (saved) return { ...base, ...JSON.parse(saved) };
  } catch {
    // нет доступа к хранилищу — начинаем с пустого бланка
  }
  return base;
}

export function useBlankForm() {
  const [form, setForm] = useState<BlankInput>(initialState);
  useEffect(() => {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(form));
    } catch {
      // ignore
    }
  }, [form]);
  return [form, setForm] as const;
}

import { toast } from "../components/Toast";

const CACHE = "blank-taho-v1";
const FLAG = "timesheets-offline-ready";
const SKIP = ["/app-api/", "/__g3/live", "/__g3/session", "/__g3/refresh"];

// Подключает работу без интернета: после первой загрузки всё нужное
// (страница, скрипты, значки) сохраняется на телефоне.
export function enableOffline() {
  if (!("serviceWorker" in navigator) || !("caches" in window)) return;

  const run = async () => {
    try {
      await navigator.serviceWorker.register("/app-api/sw.js", { scope: "/" });
      await navigator.serviceWorker.ready;
      await new Promise((r) => setTimeout(r, 2500)); // даём догрузиться остальному

      const cache = await caches.open(CACHE);
      const urls = new Set<string>(["/manifest.webmanifest"]);
      for (const e of performance.getEntriesByType("resource")) {
        const u = new URL(e.name);
        if (u.origin === location.origin && SKIP.some((p) => u.pathname.startsWith(p))) continue;
        if (u.protocol === "http:" || u.protocol === "https:") urls.add(e.name);
      }
      await Promise.all(
        [...urls].map(async (u) => {
          try {
            if (await cache.match(u)) return;
            if (new URL(u, location.href).origin === location.origin) {
              const res = await fetch(u);
              if (res.ok) await cache.put(u, res);
            } else {
              await cache.put(u, await fetch(u, { mode: "no-cors" }));
            }
          } catch {
            // не удалось сохранить один файл — остальные всё равно сохранятся
          }
        }),
      );

      // сама страница с приложением
      const page = await fetch(location.pathname, { headers: { accept: "text/html" } });
      if (page.ok && !page.redirected) {
        const text = await page.text();
        if (text.includes("mountGlideApp")) {
          await cache.put("/shell", new Response(text, { headers: { "content-type": "text/html; charset=utf-8" } }));
          if (!localStorage.getItem(FLAG)) {
            localStorage.setItem(FLAG, "1");
            toast.success("Приложение готово работать без интернета");
          }
        }
      }
    } catch {
      // без service worker приложение работает как обычно, только с интернетом
    }
  };

  if (document.readyState === "complete") run();
  else window.addEventListener("load", run, { once: true });
}
// ПРИМЕЧАНИЕ: проверка "mountGlideApp" относится к платформе Glide. В своём проекте
// замените её на любую метку вашей HTML-страницы, например "id=\"root\"".

      <div className="grid grid-cols-2 gap-3 md:grid-cols-3">
        <Stat label="Вождение за месяц" value={unitHM(result.driveU)} />
        <Stat label="Пробег за месяц" value={result.km ? `${result.km} км` : "—"} />
        <Stat
          label="Замечания"
          value={String(result.warningCount)}
          context={result.warningCount ? "нужно проверить" : "всё по правилам"}
        />
      </div>

      {issues.length > 0 && (
        <Card>
          <h3 className="text-base font-semibold">Что проверить</h3>
          <ul className="mt-2 space-y-1 text-sm text-secondary">
            {issues.map((d) => (
              <li key={d.iso}>
                {d.dateLabel}: {d.warnings.join("; ")}
              </li>
            ))}
          </ul>
        </Card>
      )}

      <section className="space-y-3">
        <div className="flex items-center justify-between gap-2">
          <h3 className="text-base font-semibold">Лист {current + 1} из {pages}</h3>
          <div className="flex gap-2">
            <Button variant="secondary" size="icon" aria-label="Предыдущий лист" disabled={current === 0} onClick={() => setPage(current - 1)}>
              <ChevronLeft />
            </Button>
            <Button variant="secondary" size="icon" aria-label="Следующий лист" disabled={current >= pages - 1} onClick={() => setPage(current + 1)}>
              <ChevronRight />
            </Button>
          </div>
        </div>
        <div className="mx-auto max-w-3xl overflow-hidden rounded-lg border border-border bg-raised">
          <div ref={host} />
        </div>
      </section>

      <section className="space-y-3">
        <h3 className="text-base font-semibold">Дни месяца</h3>
        <DataTable columns={COLUMNS} data={result.days} />
      </section>
    </div>
  );
}

const CACHE = "blank-taho-v1";
const SKIP = ["/app-api/", "/__g3/live", "/__g3/session", "/__g3/refresh"];

self.addEventListener("install", () => self.skipWaiting());
self.addEventListener("activate", (e) =>
  e.waitUntil(
    (async () => {
      for (const k of await caches.keys()) if (k !== CACHE) await caches.delete(k);
      await self.clients.claim();
    })(),
  ),
);

self.addEventListener("fetch", (e) => {
  const req = e.request;
  if (req.method !== "GET") return;
  const url = new URL(req.url);
  if (url.protocol !== "http:" && url.protocol !== "https:") return;
  if (url.origin === self.location.origin && SKIP.some((p) => url.pathname.startsWith(p))) return;
  e.respondWith(req.mode === "navigate" ? page(req) : asset(req, e));
});

// Страница: сначала сеть, без сети — последняя сохранённая.
async function page(req) {
  const cache = await caches.open(CACHE);
  try {
    const res = await fetch(req);
    if (res.ok && !res.redirected) {
      const text = await res.clone().text();
      if (text.includes("mountGlideApp")) await cache.put("/shell", new Response(text, { headers: { "content-type": "text/html; charset=utf-8" } }));
    }
    return res;
  } catch (err) {
    return (await cache.match("/shell")) || Response.error();
  }
}

// Остальное: сразу из памяти, а свежее подтягивается в фоне.
async function asset(req, e) {
  const cache = await caches.open(CACHE);
  const hit = await cache.match(req);
  const net = fetch(req)
    .then((res) => {
      if (res && (res.ok || res.type === "opaque")) cache.put(req, res.clone());
      return res;
    })
    .catch(() => null);
  if (hit) {
    e.waitUntil(net);
    return hit;
  }
  return (await net) || Response.error();
}
